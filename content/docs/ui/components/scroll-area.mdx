## Usage

```tsx
import { ScrollArea } from '@/components/ui/scroll-area'

export function Example() {
  return (
    <ScrollArea className="h-[200px] w-[350px] rounded-md border p-4">
      Long content goes here…
    </ScrollArea>
  )
}
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `maskHeight` | `number` | `30` | Edge mask size in px. Pass `0` to disable. |
| `maskClassName` | `string` | — | Classes for mask overlays |
| `viewportClassName` | `string` | — | Classes for the viewport |
| `scrollHideDelay` | `number` | `0` | Desktop scrollbar hide delay (ms) |
