## Usage

```tsx
import { JsonExplorer } from '@/components/ui/json-explorer'

export function Example() {
  return (
    <JsonExplorer
      data={{ hello: 'world', count: 3 }}
      title="Payload"
    />
  )
}
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `data` | `JsonValue` | — | JSON value to explore |
| `rootName` | `string` | `"data"` | Path prefix for the root node |
| `defaultDepth` | `number` | `1` | Initial expand depth |
| `defaultExpanded` | `string[]` | — | Paths expanded on mount |
| `title` | `string` | — | Header label |
| `onCopyPath` | `(path: string) => void` | — | Called after a path is copied |
| `className` | `string` | — | Root class name |
